{%- comment -%} Renders an expandable BibTeX drawer for a single paper. Usage: {% include paper_bibtex.html link=link type="inproceedings" %} - link : a paper entry from _data/conference.yml or _data/preprint.yml - type : "inproceedings" (published) or anything else (preprint / arXiv) If the entry defines a `bibtex` field, it is used verbatim; otherwise a clean BibTeX entry is auto-generated from title / authors / year. {%- endcomment -%} {%- assign ob = '{' -%} {%- assign cb = '}' -%} {%- assign entrytype = "article" -%} {%- if include.type == "inproceedings" -%}{%- assign entrytype = "inproceedings" -%}{%- endif -%} {%- comment -%} Clean authors: drop HTML, drop the 【date】 tag, split on commas {%- endcomment -%} {%- assign authors_src = include.link.authors | strip_html | split: "【" | first | strip -%} {%- assign author_parts = authors_src | split: "," -%} {%- assign authors_bib = "" -%} {%- for a in author_parts -%} {%- assign an = a | strip | replace: "*", "" | replace: ".", "" | strip -%} {%- if an != "" -%} {%- if authors_bib == "" -%}{%- assign authors_bib = an -%} {%- else -%}{%- assign authors_bib = authors_bib | append: " and " | append: an -%}{%- endif -%} {%- endif -%} {%- endfor -%} {%- comment -%} Year: prefer conference year for published, else the 【date】 year {%- endcomment -%} {%- assign year = "" -%} {%- if include.link.authors contains "【" -%} {%- assign yraw = include.link.authors | split: "【" | last | split: "】" | first -%} {%- assign year = yraw | split: "." | first | strip -%} {%- endif -%} {%- assign pubyear = year -%} {%- if entrytype == "inproceedings" and include.link.conference_short -%} {%- assign cs_words = include.link.conference_short | split: " " -%} {%- for w in cs_words -%} {%- assign ws = w | strip -%} {%- if ws.size == 4 and ws contains "20" -%}{%- assign pubyear = ws -%}{%- endif -%} {%- endfor -%} {%- endif -%} {%- if pubyear == "" -%}{%- assign pubyear = "" -%}{%- endif -%} {%- comment -%} Citation key: firstauthorsurname + year + firsttitleword {%- endcomment -%} {%- assign first_author = author_parts | first | strip | replace: "*", "" | replace: ".", "" | strip -%} {%- assign fa_words = first_author | split: " " -%} {%- assign surname = fa_words | last | downcase -%} {%- assign title_words = include.link.title | split: " " -%} {%- assign title_first = title_words | first | downcase | replace: ":", "" | replace: ",", "" | replace: "?", "" -%} {%- assign bibkey = surname | append: pubyear | append: title_first -%} {%- comment -%} arXiv id from the pdf/abs link {%- endcomment -%} {%- assign arxiv_id = "" -%} {%- if include.link.arxiv contains "arxiv.org" -%} {%- assign arxiv_id = include.link.arxiv | split: "/" | last | replace: ".pdf", "" -%} {%- endif -%}
